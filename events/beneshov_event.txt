# Czechoslovakia Receives Beneshov Demand for Moravia
country_event = {
	id = benesov.20
	title = benesov.20.t
	desc = benesov.20.d
	picture = GFX_report_event_generic_conference
	is_triggered_only = yes

	option = {
		name = benesov.20.a
		ai_chance = { base = 100 }
		if = {
			limit = { 75 = { is_owned_by = ROOT } }
			75 = { transfer_state_to = FROM }
		}
		set_country_flag = cze_conceded_moravia_to_ben
		add_opinion_modifier = {
			target = FROM
			modifier = ben_cze_conceded_moravia
		}
		FROM = {
			country_event = { id = benesov.21 days = 1 }
		}
	}

	option = {
		name = benesov.20.b
		ai_chance = { base = 0 }
		if = {
			limit = {
				75 = {
					is_owned_by = ROOT
					NOT = { is_claimed_by = FROM }
				}
			}
			75 = { add_claim_by = FROM }
		}
		set_country_flag = cze_rejected_moravia_to_ben
		add_opinion_modifier = {
			target = FROM
			modifier = ben_cze_rejected_moravia
		}
		FROM = {
			country_event = { id = benesov.22 days = 1 }
		}
	}
}

	id = benesov.22
	title = benesov.22.t
	desc = benesov.22.d
	picture = GFX_report_event_BEN_adam_war_order
	is_triggered_only = yes

	option = {
		name = benesov.22.a
		set_country_flag = ben_moravia_demands_rejected
		if = {
			limit = {
				country_exists = CZE
				NOT = { has_war_with = CZE }
				NOT = { has_wargoal_against = { target = CZE } }
			}
			create_wargoal = {
				type = take_claimed_state
				target = CZE
				generator = { 75 }
			}
		}
		add_war_support = 0.05
	}
}
