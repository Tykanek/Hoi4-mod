add_namespace = prague_event



country_event = { #Default template
	id = prague.1
	title = "prague.1.t"
	desc = "prague.1.d"
	picture = GFX_report_event_generic_read_write

	is_triggered_only = yes #yes = only triggered by script, no = can be triggered by 											script or by chance

	mean_time = { #Mean time between event occurrences
		days = 1
	}

	fire_only_once = yes #yes = event can only fire once, no = event can fire multiple 										times

	option = {
		name = "prague.1.a" #The first option the player can choose
		if = {
			limit = { #If the condition is met
				has_war = yes
			}
			every_other_country = { #For each other country that is in war with you
				limit = {
					is_in_faction_with = ROOT
					is_at_war = yes
				}
				add_political_power = 50
				add_ideas = prague_event_idea #adds the national idea to the country that 										is in a faction with the ROOT country and is at war
			}

		}
		else = { #If the condition is not met
			add_political_power = -50
		}
	}
	option = {
		name = "prague.1.b" #The second option the player can choose
		if = {
			limit = { #If the condition is met
				has_war = yes
			}
			every_other_country = { #For each other country
				limit = {
					is_in_faction_with = ROOT
					is_at_war = yes
				}
				add_political_power = -50
				remove_ideas = prague_event_idea #removes the national idea from the 							country thatis in a faction with the ROOT country and is at war
			}

		}
		else = { #If the condition is not met
			add_political_power = 50
		}
}
}
