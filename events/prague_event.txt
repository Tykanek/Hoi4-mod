# ================================================================
# STAGED CZECH DEMANDS
# ================================================================

country_event = {
    id = pra_mohammedis.18
    title = pra_mohammedis.18.t
    desc = pra_mohammedis.18.d
    is_triggered_only = yes

    option = {
        name = pra_mohammedis.18.a
        ai_chance = { base = 99 }
        if = {
            limit = { 1088 = { is_owned_by = CZE } }
            1088 = { transfer_state_to = PRA }
        }
        set_country_flag = cze_conceded_prague_hinterland
        add_opinion_modifier = {
            target = PRA
            modifier = pra_cze_conceded_prague_demand
        }
        PRA = {
            set_country_flag = pra_hinterland_demand_accepted
            add_political_power = 35
            add_stability = 0.02
        }
    }

    option = {
        name = pra_mohammedis.18.b
        ai_chance = { base = 1 }
        if = {
            limit = {
                1088 = {
                    is_owned_by = CZE
                    NOT = { is_claimed_by = PRA }
                }
            }
            1088 = { add_claim_by = PRA }
        }
        set_country_flag = cze_rejected_prague_hinterland
        add_opinion_modifier = {
            target = PRA
            modifier = pra_cze_rejected_prague_demand
        }
        PRA = {
            set_country_flag = pra_hinterland_demand_refused
            if = {
                limit = {
                    1088 = { is_owned_by = CZE }
                }
                create_wargoal = { type = take_claimed_state target = CZE }
            }
            add_war_support = 0.03
        }
    }
}
