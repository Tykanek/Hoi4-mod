pra_ben_resolve_bidding_war_effect = {
	# PRA wins: Both exist and PRA offered more, OR PRA exists and BEN does not
	if = {
		limit = {
			country_exists = PRA
			OR = {
				NOT = { country_exists = BEN }
				check_variable = { var = global.pra_ben_bidding_balance value = 0 compare = greater_than }
			}
		}
		if = {
			limit = {
				PRA = { has_character = PRA_BEN_shared_field_marshal }
			}
			PRA = {
				PRA_BEN_shared_field_marshal = {
					set_character_flag = pra_ben_elbl_contract_signed
				}
			}
		}
		else_if = {
			limit = {
				BEN = { has_character = PRA_BEN_shared_field_marshal }
			}
			BEN = {
				PRA_BEN_shared_field_marshal = {
					set_character_flag = pra_ben_elbl_contract_signed
					set_nationality = PRA
				}
			}
			PRA = {
				PRA_BEN_shared_field_marshal = {
					set_character_flag = pra_ben_elbl_contract_signed
				}
			}
		}
		PRA = {
			country_event = { id = pra_ben_bidding_war.1 }
		}
		if = {
			limit = { country_exists = BEN }
			BEN = { country_event = { id = pra_ben_bidding_war.2 } }
		}
	}

	# BEN wins: Both exist and BEN offered more, OR BEN exists and PRA does not
	if = {
		limit = {
			country_exists = BEN
			OR = {
				NOT = { country_exists = PRA }
				check_variable = { var = global.pra_ben_bidding_balance value = 0 compare = less_than }
			}
		}
		if = {
			limit = {
				PRA = { has_character = PRA_BEN_shared_field_marshal }
			}
			PRA = {
				PRA_BEN_shared_field_marshal = {
					set_character_flag = pra_ben_elbl_contract_signed
					set_nationality = BEN
				}
			}
			BEN = {
				PRA_BEN_shared_field_marshal = {
					set_character_flag = pra_ben_elbl_contract_signed
				}
			}
		}
		else_if = {
			limit = {
				BEN = { has_character = PRA_BEN_shared_field_marshal }
			}
			BEN = {
				PRA_BEN_shared_field_marshal = {
					set_character_flag = pra_ben_elbl_contract_signed
				}
			}
		}
		BEN = {
			country_event = { id = pra_ben_bidding_war.3 }
		}
		if = {
			limit = { country_exists = PRA }
			PRA = { country_event = { id = pra_ben_bidding_war.4 } }
		}
	}

	# Tie: Both exist and offers are equal
	if = {
		limit = {
			country_exists = PRA
			country_exists = BEN
			check_variable = { var = global.pra_ben_bidding_balance value = 0 compare = equals }
		}
		if = {
			limit = {
				PRA = { has_character = PRA_BEN_shared_field_marshal }
			}
			PRA = {
				PRA_BEN_shared_field_marshal = {
					retire = yes
				}
			}
		}
		else_if = {
			limit = {
				BEN = { has_character = PRA_BEN_shared_field_marshal }
			}
			BEN = {
				PRA_BEN_shared_field_marshal = {
					retire = yes
				}
			}
		}
		PRA = { country_event = { id = pra_ben_bidding_war.5 } }
		BEN = { country_event = { id = pra_ben_bidding_war.5 } }
	}

	# Both ceased to exist
	if = {
		limit = {
			NOT = { country_exists = PRA }
			NOT = { country_exists = BEN }
		}
		if = {
			limit = {
				PRA = { has_character = PRA_BEN_shared_field_marshal }
			}
			PRA = {
				PRA_BEN_shared_field_marshal = {
					retire = yes
				}
			}
		}
		else_if = {
			limit = {
				BEN = { has_character = PRA_BEN_shared_field_marshal }
			}
			BEN = {
				PRA_BEN_shared_field_marshal = {
					retire = yes
				}
			}
		}
	}
}
