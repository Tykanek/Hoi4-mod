# Universal leader trait synchronization for completed doping research.
# Any country that has researched a doping technology automatically transfers
# the corresponding trait to its current political leader.
# Each technology is evaluated independently with explicit trait-existence guards
# to prevent duplicate trait assignment warnings.
GCA_sync_doping_leader_traits = {
	# Steroids I — Natural Transformation (+1% Division Attack)
	if = {
		limit = {
			has_tech = GCA_steroids_natural_transformation_tech
			NOT = { has_country_leader_with_trait = GCA_steroids_natural_transformation_trait }
		}
		add_country_leader_trait = GCA_steroids_natural_transformation_trait
	}

	# Doping II — Enhanced Performance (+1% Division Defense)
	if = {
		limit = {
			has_tech = GCA_doping_enhanced_performance_tech
			NOT = { has_country_leader_with_trait = GCA_doping_enhanced_performance_trait }
		}
		add_country_leader_trait = GCA_doping_enhanced_performance_trait
	}

	# Doping III — Grznar's Final Form (+1% Division Attack)
	if = {
		limit = {
			has_tech = GCA_grznar_final_form_tech
			NOT = { has_country_leader_with_trait = GCA_grznar_final_form_trait }
		}
		add_country_leader_trait = GCA_grznar_final_form_trait
	}
}
